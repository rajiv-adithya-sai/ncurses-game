#include<stdio.h>
#include <ncurses.h>
#include<string.h>
//#include<locale.h>
#define dim 20
#define init_posn 5

char index[dim][dim];
int point_row = init_posn;
void init_title(){
	for(int i=0;i<dim;i++){
		for(int j =0;j<dim;j++){
			index[i][j] = ' ';
		}
	}
	FILE* fin = fopen("assets/title.txt" , "r");
	char temp[100];
	int row = 0;
	while(fgets(fin , "%s" , temp)){
		temp[strcspn(temp , "\n")] = '\0';
		for(int j=0;j<strlen(temp);j++){
			index[row][j] = temp[j];
		}
		row++;
	}
	fclose(fin);
}

void option_select(){
	int ch = getch();
	if(ch == KEY_DOWN){
		point_row ++;
	}
	else if (ch == KEY_UP) {
		point_row--;
	}
}

void draw_title(){
	for(int i=0;i<dim;i++){
		for(int j=0;j<dim;j++){
			if(i == point_row){
				mvaddch(i+4 , j , index[i][j]);
			}
			
			else mvaddch(i+4 ,j, index[i][j]);
		}	
	}
}
